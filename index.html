PYTHON · REFERENCE REQUEST
import requests

with open("ar_encoded.jpg", "rb") as image:
    response = requests.post(
        "https://api.framedge.com/detect_and_decode",
        files={"file": (
            "ar_encoded.jpg", image, "image/jpeg"
        )},
    )
print(response.text)
